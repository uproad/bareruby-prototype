struct Servo
    pwm: struct bareruby_pwm_t
    angle: int32

function Servo_initialize(self: struct Servo*, pin: int32) -> void
    declare temporary_1: struct bareruby_pwm_t
    bareruby_pwm_init(&temporary_1, pin, 50, 0)
    assign self.pwm = temporary_1
    assign self.angle = 0
    return

function Servo_move(self: struct Servo*, angle: int32) -> void
    assign self.angle = angle
    bareruby_pwm_pulse_width_us(&self.pwm, (1000 + ((angle * 1000) / 180)))
    return

function bareruby_main() -> void
    declare temporary_1: struct Servo
    Servo_initialize(&temporary_1, 15)
    declare servo: struct Servo = temporary_1
    while (true)
        declare temporary_2: int32 = 180
        for (declare angle: int32 = 0; (angle <= temporary_2); assign angle = (angle + 1))
            Servo_move(&servo, angle)
            bareruby_machine_delay_us(500)
        bareruby_sleep_ms(300)
